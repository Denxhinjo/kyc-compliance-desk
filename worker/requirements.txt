# Postgres driver. psycopg 3 (not psycopg2 — that is maintenance-only now).
# [binary] ships precompiled wheels so there is no C toolchain needed.
#
# Pinned to the same version as the root requirements.txt, which the Vercel
# function uses. 3.2.13 rather than 3.2.3 because Vercel builds on CPython
# 3.14 and the first cp314 wheels appeared in 3.2.10 — and a function running a
# different psycopg from the worker is a difference that only shows up in
# production.
psycopg[binary]==3.2.13

# Reads the single root .env so the worker uses the same configuration as
# /web and docker-compose.
python-dotenv==1.0.1

# Tests. The lifecycle state machine is pure functions and is the safety-
# critical part of Phase 4, so it is tested in isolation rather than only
# through the database.
pytest==8.3.4

# Fuzzy name matching. C-backed, so a 20,000-entry list is searchable per
# applicant without a separate search service.
rapidfuzz==3.14.6
